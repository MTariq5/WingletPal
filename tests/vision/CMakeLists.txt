cmake_minimum_required(VERSION 3.16)
project(FaceTrackerTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Test)
find_package(OpenCV 4 REQUIRED COMPONENTS core)
enable_testing()
add_executable(face_tracker_tests FaceTrackerTests.cpp ../../vision/FaceTracker.cpp)
target_include_directories(face_tracker_tests PRIVATE ../../vision ${OpenCV_INCLUDE_DIRS})
target_link_libraries(face_tracker_tests PRIVATE Qt6::Core Qt6::Test ${OpenCV_LIBS})
add_test(NAME face_tracker_tests COMMAND face_tracker_tests)
