cmake_minimum_required(VERSION 3.16)
project(LocationTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Test Network)
enable_testing()
add_executable(location_tests LocationTests.cpp ../../LocationProvider.cpp)
target_include_directories(location_tests PRIVATE ../..)
target_link_libraries(location_tests PRIVATE Qt6::Core Qt6::Test)
add_test(NAME location_tests COMMAND location_tests)

add_executable(city_search_tests CitySearchTests.cpp ../../CitySearch.cpp ../../LocationProvider.cpp)
target_include_directories(city_search_tests PRIVATE ../..)
target_link_libraries(city_search_tests PRIVATE Qt6::Core Qt6::Test Qt6::Network)
add_test(NAME city_search_tests COMMAND city_search_tests)
