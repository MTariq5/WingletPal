cmake_minimum_required(VERSION 3.16)
project(FocusSessionTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Test)
enable_testing()
add_executable(focus_session_tests FocusSessionTests.cpp ../../FocusSession.cpp)
target_include_directories(focus_session_tests PRIVATE ../..)
target_link_libraries(focus_session_tests PRIVATE Qt6::Core Qt6::Test)
add_test(NAME focus_session_tests COMMAND focus_session_tests)
